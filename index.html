const CACHE = 'budget-v2'; // Обновили версию кэша self.addEventListener('install', e => { self.skipWaiting(); // Активировать нового SW сразу }); self.addEventListener('activate', e => { e.waitUntil( caches.keys().then(keys => { return Promise.all( keys.map(key => { if (key !== CACHE) { return caches.delete(key); // Удаляем старые кэши } }) ); }) ); }); self.addEventListener('fetch', e => { // Пропускаем запросы к Firebase if (e.request.url.includes('firebase') || e.request.url.includes('googleapis')) return; // Стратегия "Сначала сеть, потом кэш" для HTML и JS // Это гарантирует, что вы увидите свежие обновления интерфейса if (e.request.mode === 'navigate' || e.request.headers.get('accept').includes('text/html')) { e.respondWith( fetch(e.request) .then(res => { const resClone = res.clone(); caches.open(CACHE).then(c => c.put(e.request, resClone)); return res; }) .catch(() => caches.match(e.request)) ); return; } // Для остальных файлов — "Сначала кэш, потом сеть" e.respondWith( caches.match(e.request).then(res => res || fetch(e.request)) ); });
